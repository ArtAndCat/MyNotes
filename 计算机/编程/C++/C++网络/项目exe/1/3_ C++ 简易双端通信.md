# 客户端

```cpp
#include <boost/asio.hpp>
#include <iostream>
#include <string>

int main() {
    try {
        boost::asio::io_context io;

        // 1. 解析地址
        boost::asio::ip::tcp::resolver resolver(io);
        auto endpoints = resolver.resolve("127.0.0.1", "12345");

        // 2. 创建 socket 并连接
        boost::asio::ip::tcp::socket socket(io);
        boost::asio::connect(socket, endpoints);
        std::cout << "Connected to server." << std::endl;

        // 3. 发送消息
        std::string message = "Hello, Boost.Asio!";
        boost::asio::write(socket, boost::asio::buffer(message));
        std::cout << "Sent: " << message << std::endl;

        // 4. 接收响应
        char reply[1024];
        boost::system::error_code ec;
        size_t len = socket.read_some(boost::asio::buffer(reply), ec);

        if (ec) {
            std::cerr << "Read error: " << ec.message() << std::endl;
            return 1;
        }

        std::cout << "Received: " << std::string(reply, len) << std::endl;
    } catch (std::exception& e) {
        std::cerr << "Exception: " << e.what() << std::endl;
        return 1;
    }
    return 0;
}
```



# 服务端  (记得根据情况改ip地址 端口记得保持一致)


```cpp
#include <boost/asio.hpp>
#include <iostream>
#include <string>

int main() {
    try {
        boost::asio::io_context io;

        // 1. 创建 acceptor，监听 12345 端口
        boost::asio::ip::tcp::acceptor acceptor(io,
            boost::asio::ip::tcp::endpoint(boost::asio::ip::tcp::v4(), 12345));

        std::cout << "Server listening on port 12345..." << std::endl;

        // 2. 接受一个客户端连接（阻塞）
        boost::asio::ip::tcp::socket socket(io);
        acceptor.accept(socket);
        std::cout << "Client connected." << std::endl;

        // 3. 读取客户端数据
        char data[1024];
        boost::system::error_code ec;
        size_t len = socket.read_some(boost::asio::buffer(data), ec);

        if (ec == boost::asio::error::eof) {
            std::cout << "Client closed connection before sending data." << std::endl;
            return 0;
        } else if (ec) {
            throw boost::system::system_error(ec);
        }

        std::cout << "Received: " << std::string(data, len) << std::endl;

        // 4. 将数据原样发回
        boost::asio::write(socket, boost::asio::buffer(data, len));
        std::cout << "Echoed back." << std::endl;

        // 5. 关闭连接（socket 析构时会自动关闭）
    } catch (std::exception& e) {
        std::cerr << "Exception: " << e.what() << std::endl;
        return 1;
    }
    return 0;
}
```
